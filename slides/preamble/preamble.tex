% Page-relative coordinate: (x,y) as fractions of the page from the top-left corner.
\newcommand{\PG}[2]{([shift={(#1\paperwidth,-#2\paperheight)}]current page.north west)}

% ---- Bottom band ----
\newlength{\bottombandheight}
\setlength{\bottombandheight}{0.125\paperheight}

\setbeamertemplate{background}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\the\paperwidth,\the\paperheight);
    \fill[UniversityBlue] (0,0) rectangle (\the\paperwidth,\bottombandheight);
  \end{tikzpicture}%
}

\setbeamertemplate{footline}{%
  \vbox to \bottombandheight{%
    \vfill
    \hbox to \paperwidth{%
      \kern0.005\paperwidth
      \raisebox{-0.40\height}[0pt][0pt]{\uofglogo{0.15\paperwidth}}%
      \hfill
      \usebeamerfont{footline}\color{white}%
      \ifnum\insertframenumber>0 \insertframenumber\fi
      \kern0.02\paperwidth}%
    \vskip0.36\bottombandheight
  }%
}
