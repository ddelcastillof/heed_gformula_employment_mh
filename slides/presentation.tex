% Introduction
% =====================================
\begin{frame}{Background}
  \begin{itemize}
    \item Employment is a major determinant of mental health and a key driver of social inequalities\cite{Marmot2005-kx,Nie2020-zw}.
    \item Re-employment has restorative effects on mental health. However, the duration of the lingering effects of unemployment on mental health is unclear\cite{Luhmann2012-em,Lahelma1989-we}.
    \item Studying the effect of unemployment trajectories on mental health is complicated dues to the presence of causal structures such as time-varying confounding and treatment-confounder feedback\cite{Bijlsma2019-hc}.
  \end{itemize}
\end{frame}

% =====================================
% Methods
% =====================================
\begin{frame}[shrink=10]{Methods}
  \begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}
    \begin{itemize}
      \item Data from January 2010 — May 2020 from the UK Household Longitudinal Study (UKHLS)\cite{ukhls6614}.
      \item \boldtxt{Eligibility:} Working-age adults (25 to 65) who were employed at baseline (n = 21,104). Students, early retired, or long-term sick at baseline were excluded.
      \item \boldtxt{Variables:}
        \begin{itemize}
          \item \boldtxt{Exposure:} Employment status (employed, unemployed).
          \item \boldtxt{Outcome:} Mental health status (MCS SF-12).
          \item \boldtxt{Covariates:} gender, ethnicity, education, age, location (GOR), marital status, household income (log), house tenancy, number of dependants, receipt of welfare benefits, physical health status (PCS SF-12).
        \end{itemize}
    \end{itemize}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}
  \begin{itemize}
    \item \boldtxt{Causal estimand:} Marginal mean (mental health score) under different employment trajectories.
    \item We used the g-formula (\pkg{gFormulaMI}) to estimate the effect of unemployment trajectories on mental health (\texttt{R 4.5.2})\cite{Bartlett2025-iw}.
    \begin{itemize}
      \item We used multiple imputation (\pkg{mice}) to account for missing data (m=50, i=15).
      \item We simulated potential outcomes under different employment trajectories, e.g. (E-E-E-E vs U-U-U-U).
    \end{itemize}
  \end{itemize}
  \end{minipage}
\end{frame}

% =====================================
% Results
% =====================================
\begin{frame}{Results (i)}{Sample characteristics}
  \begin{itemize}
    \item The mean age of the sample was 44.7 years (SD = 10).
    \item Women made up 56.4\% of the sample.
    \item The most frequent trajectory was E-E-E-E (78.7\%), and the least was E-U-E-U (0.25\%).
  \end{itemize}
\end{frame}

% =====================================
% Conclusions
% =====================================
\begin{frame}{Conclusions}
  \begin{itemize}
    \item Employment trajectories have substantial adverse mental health effects, with more unemployment episodes having greater effects.
    \item Unemployment effects are greatest within the first year, they persist long-term and vary across trajectories.
    \item \boldtxt{Potential limitations:} potential residual confounding, differential attrition, inability to differentiate detailed employment status categories.
  \end{itemize}
\end{frame}
