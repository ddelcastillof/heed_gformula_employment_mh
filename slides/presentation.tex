% Introduction
% =====================================
\begin{frame}{Background}
  \begin{itemize}
    \item Employment is a major determinant of mental health and a key driver of social inequalities\cite{Marmot2005-kx,Nie2020-zw}.
    \item Re-employment has restorative effects on mental health. However, the duration of the lingering effects of unemployment on mental health is unclear\cite{Luhmann2012-em,Lahelma1989-we}.
    \item Studying the effect of unemployment trajectories on mental health is complicated by the presence of causal structures such as time-varying confounding and treatment-confounder feedback\cite{Bijlsma2019-hc}.
  \end{itemize}
\end{frame}

% =====================================
% Methods
% =====================================
\begin{frame}{Methods}
  \begin{minipage}[t]{0.55\textwidth}
    \vspace{0pt}
    \footnotesize
    \begin{itemize}
      \item Data from January 2011 - May 2020 from the UK Household Longitudinal Study (UKHLS)\cite{ukhls6614}.
      \item \boldtxt{Eligibility:} Working-age adults (25 to 64) who were employed at baseline. Students, early retired, or long-term sick at baseline were excluded.
    \end{itemize}
  \boldtxt{Variables:}
  \begin{itemize}
    \item \boldtxt{Exposure:} Employment status (employed, unemployed).
    \item \boldtxt{Outcome:} SF-12 mental health component score (SF-12 MCS).
    \item \boldtxt{Covariates:} sex, ethnicity, education, age, location (GOR), marital status, household income, house tenancy, dependants, welfare benefits, physical health status (SF-12 PCS).
  \end{itemize}
  \end{minipage}\hfill

    \vspace{0pt}
    \centering
    \impute[0.90]
  \end{minipage}
\end{frame}

% =====================================
% Conclusions
% =====================================
\begin{frame}{Conclusions}
  \begin{itemize}
    \item The causal effect of unstable employment on mental health lands between -0.7 and -3.5 points for the SF-12 MCS scale.
    \item Unstable employment have substantial adverse mental health effects, with more unemployment episodes having greater effects.
 %   \item \boldtxt{Potential limitations:} potential residual confounding, differential attrition, inability to differentiate detailed employment status categories.
    \item Unstable employment leaves `scars' on mental health. 
    \item Policymakers should prioritise efforts to secure and maintain stable employment opportunities through employability and job retention schemes.
  \end{itemize}
\end{frame}
